cmake_minimum_required(VERSION 3.12)
project(g1_interface_python LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(unitree_sdk2 REQUIRED)
find_package(pybind11 REQUIRED)
pybind11_add_module(g1_interface g1_interface.cpp python_bindings.cpp)
target_link_libraries(g1_interface PRIVATE unitree_sdk2)
install(TARGETS g1_interface DESTINATION .)
